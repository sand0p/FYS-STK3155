Here’s a complete, working example for torchtext 0.6.0:

```python
import torch
from torch.utils.data import DataLoader
from torchtext.datasets import text_classification

# Check available datasets (optional, helps verify the key)
print("Available datasets:", list(text_classification.DATASETS.keys()))
# You should see 'AmazonReviewFull' among the keys

# Load train and test datasets (ngrams=1 is common; you can try 2 or 3 as well)
NGRAMS = 1
train_dataset, test_dataset = text_classification.DATASETS['AmazonReviewFull'](
    root='.data',  # cache/download directory
    ngrams=NGRAMS
)

print(f"Train size: {len(train_dataset)} | Test size: {len(test_dataset)}")

# Inspect one sample: (label, text_tensor)
label, text_tensor = train_dataset[0]
print("Label:", label)                 # integer in [1, 5]
print("Vectorized text shape:", text_tensor.shape)
print("First few token IDs:", text_tensor[:10])

# Collate function for bag-of-ngrams datasets
# This packs variable-length tensors into a single flat tensor with offsets
def generate_batch(batch):
    labels = torch.tensor([entry[0] for entry in batch], dtype=torch.long)
    texts = [entry[1] for entry in batch]
    offsets = [0] + [len(t) for t in texts[:-1]]
    offsets = torch.tensor(offsets, dtype=torch.long).cumsum(dim=0)
    texts = torch.cat(texts)  # 1D tensor of token/feature ids
    return texts, offsets, labels

# DataLoader for batching
train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True, collate_fn=generate_batch)

# Iterate over one batch
for texts, offsets, labels in train_loader:
    print("texts shape:", texts.shape)     # [total_tokens_in_batch]
    print("offsets shape:", offsets.shape) # [batch_size]
    print("labels shape:", labels.shape)   # [batch_size]
    break
```

- There is no split='train' argument. Use text_classification.DATASETS['AmazonReviewFull'](...) which returns (train_dataset, test_dataset).
- Examples are numericalized tensors (bag-of-ngrams), not raw text strings.
- Use a collate function (like generate_batch above) with offsets for efficient embedding/Bag-of-ngrams models.
